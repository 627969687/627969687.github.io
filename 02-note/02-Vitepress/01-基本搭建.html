2.1 搭建github仓库 ​

  1. 搭建图床仓库
  2. 搭建父子仓库
    1. github创建父仓库，设置为public，设置token
    2. github创建子仓库，设置为private，设置token
    3. 拉取到本地
  3. 搭建图床仓库
yml
# 拉取父仓库
git clone https://username:token@github.com/username/username.github.io.git
# cd进入父仓库
cd username.github.io
# 添加子仓库（子仓库不能用token的形式连接，因为构建子模块会生成子模块配置.gitmodules提交到github，github不允许token被展示）
git submodule add -f https://github.com/username/子仓库名.git 子仓库存放的相对路径名
# 推送
git push

4. 博客设置 ​

4.1 配置 ​

修改.vitepress/config.mts文件

js
import { defineConfig } from 'vitepress'

export default defineConfig({
  title: "博客名称",
  description: "博客描述",
  lang: 'zh-CN',
  lastUpdated: true, // 文章最后更新时间
  srcDir: 'markdown', // 笔记扫描路径，这里我的子仓库文件夹命名为markdown
  markdown: {
    lineNumbers: true, // 显示md代码行数
  },
  themeConfig: {
	logo: '博客logo',
	// 目录配置
    outline: {
      level: [1,2,3], // 根据md标题等级生成目录，这里设置了1～3级
      label: '目录',
    },
    socialLinks: [
      { icon: 'github', link: 'github地址' }
    ],
	// 导航栏配置
    nav: [
      { text: '标题1', link: '这里建议链接你这个分组下的第一篇md' },
      { text: '标题2', link: '' },
    ],
    // 侧边栏配置
    sidebar: {
      // 当用户位于 `guide` 目录时，会显示此侧边栏
      '/guide/': [
        {
          text: 'Guide',
          items: [
            { text: 'Index', link: '/guide/' },
            { text: 'One', link: '/guide/one' },
            { text: 'Two', link: '/guide/two' }
          ]
        }
      ],

      // 当用户位于 `config` 目录时，会显示此侧边栏
      '/config/': [
        {
          text: 'Config',
          items: [
            { text: 'Index', link: '/config/' },
            { text: 'Three', link: '/config/three' },
            { text: 'Four', link: '/config/four' }
          ]
        }
      ]
    }
  }
})

5. 自动部署 ​

5.1 子仓库 ​

新建.github/workflows/deploy.yml文件，用于更新子仓库并告知父仓库

yml
name: Send submodule updates to parent repo

on:
  workflow_dispatch:
  push:
    branches: 
      - main

jobs:
  update:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v3
        with: 
          repository: username/username.github.io
          token: ${{ secrets.PRIVATE_TOKEN_GITHUB }} # 参考2.1搭建github仓库: github上子仓库设置的token字段
          submodules: true

      - name: Pull & update submodules recursively
        run: |
          git submodule update --init --recursive
          git submodule update --recursive --remote

      - name: Commit
        run: |
          git config user.email "username@github.com"
          git config user.name "GitHub Actions - update submodules"
          git add --all
          git commit -m "Update submodules" || echo "No changes to commit"
          git push

5.2.2 github pages ​

开启github pages

配置运行环境，vitepress生成html，将html部署到gh-pages分支

yml
name: Deploy to GitHub Pages

on:
  push:
    branches:
      - main

jobs:
  deploy:
    runs-on: ubuntu-latest
    permissions:
      contents: write

    steps:
      - uses: actions/checkout@v3
        with:
          fetch-depth: 0
          submodules: true
          token: ${{ secrets.PRIVATE_TOKEN_GITHUB }} # 父仓库token

      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '20'
          cache: yarn

      - name: Setup Pages
        uses: actions/configure-pages@v3

      - name: Install dependencies
        run: yarn install

      - name: Build the documentation
        run: yarn docs:build
		
      - name: Deploy 🚀
        uses: JamesIves/github-pages-deploy-action@v4
        with:
          token: ${{ secrets.PRIVATE_TOKEN_GITHUB }}
          folder: .vitepress/dist
