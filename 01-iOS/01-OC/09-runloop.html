1. 简介 ​

  1. RunLoop是一个保证应用程序的运行不被终止的运行循环对象
  2. RunLoop通过内部维护的事件循环来对事件/消息进行管理
  3. RunLoop不是一个简单的do...while循环，它涉及到用户态和内核态之间的切换
objc
#import <UIKit/UIKit.h>
#import "AppDelegate.h"

int main(int argc, char * argv[]) {
    @autoreleasepool {
    /* UIApplicationMain函数内部会启动主线程的`RunLoop`
	1. 处理 App 中的的各种事件（比如触摸事件、定时器事件等）
	2. 节省 CPU 资源，提高程序性能：该做事时做事，该休息时休息
    */
        return UIApplicationMain(argc, argv, nil, NSStringFromClass([AppDelegate class]));
    }
}

2. RunLoop对象 ​

2.1 基本API ​

有Foundation和Core Foundation两套API：

  1. Foundation：CFRunLoopRef的封装，面向对象的API
  2. Core Foundation：CFRunLoopRef
objc
// Foundation
[NSRunLoop mainRunLoop];     // 获取主线程的 RunLoop 对象
[NSRunLoop currentRunLoop];  // 获取当前线程的 RunLoop 对象
// Core Foundation
CFRunLoopGetMain();     // 获取主线程的 RunLoop 对象
CFRunLoopGetCurrent();  // 获取当前线程的 RunLoop 对象

2.2 本质 ​

重要的两个概念：

  1. 模式
  2. 事件源：
    1. 输入源
    2. 定时器源

RunLoop的五个类：

  1. CFRunLoopRef RunLoop对象的底层
  2. CFRunLoopModeRef 模式
  3. CFRunLoopSourceRef 输入源
  4. CFRunLoopTimerRef 定时器源
  5. CFRunLoopObserverRef

2.2.1 CFRunLoopRef ​

  1. RunLoop对象的底层就是一个CFRunLoopRef结构体
  2. 一个RunLoop对象包含多个CFRunLoopModeRef运行模式
  3. RunLoop启动时只能选择其中一个运行模式作为_currentMode
objc
typedef struct CF_BRIDGED_MUTABLE_TYPE(id) __CFRunLoop * CFRunLoopRef;
struct __CFRunLoop {
    pthread_t _pthread; // `RunLoop`与线程是一一对应关系
    CFMutableSetRef _commonModes; // 存储着 NSString 对象的集合（运行模式的名称）
    CFMutableSetRef _commonModeItems; // 存储着被标记为通用模式的`Source0`、`Source1`、`Timer`、`Observer`
    CFRunLoopModeRef _currentMode; // `RunLoop`当前的运行模式
    CFMutableSetRef _modes; // `RunLoop`所有的运行模式
};

2.2.2 CFRunLoopModeRef ​

  1. 一个CFRunLoopModeRef运行模式包含多个Source0、Source1、Timer、Observer
  2. 切换运行模式需要退出当前RunLoop后重新选择一个模式进入，切换过程不会导致程序退出
  3. 不同模式下的Source0、Source1、Timer、Observer互不影响
  4. 如果模式下没有任何Source0、Source1、Timer、Observer，会立马退出RunLoop
objc
typedef struct __CFRunLoopMode *CFRunLoopModeRef;
struct __CFRunLoopMode {
    CFStringRef _name; // mode 类型，如：NSDefaultRunLoopMode
    CFMutableSetRef _sources0; // CFRunLoopSourceRef
    CFMutableSetRef _sources1; // CFRunLoopSourceRef
    CFMutableArrayRef _observers; // CFRunLoopObserverRef
    CFMutableArrayRef _timers; // CFRunLoopTimerRef
};

ModeName作用
NSDefaultRunLoopMode默认模式，主线程默认在这个Mode下运行
UITrackingRunLoopMode界面追踪模式，用于 ScrollView 追踪触摸滑动，保证界面滑动时不受其他 Mode 影响；
UIInitializationRunLoopMode在刚启动App时第进入的第一个Mode，启动完成后就不再使用
GSEventReceiveRunLoopMode用于接受系统事件，属于内部的RunLoop模式
NSRunLoopCommonModes一个 Mode 的集合，默认包括 NSDefaultRunLoopMode 和UITrackingRunLoopMode
关于NSRunLoopCommonModes：
objc
/* 如何实现`Timer`任何模式下都可以触发？
1. 把当前`RunLoop`和`UITrackingRunLoopMode`统一加入到NSRunLoopCommonModes中
2. 将`Timer`关联到`NSRunLoopCommonModes`中
*/
NSTimer *timer = ...;
CFRunLoopAddCommonMode([[NSRunLoop currentRunLoop] getCFRunLoop],(__bridge CFStringRef) UITrackingRunLoopMode);
[[NSRunlLoop currentRunLoop] addTimer:timer forMode:NSRunLoopCommonModes];

2.2.3 CFRunLoopSourceRef ​

输入源区别
Source0添加Source0到RunLoop并不会主动唤醒线程，需要手动唤醒
1. 触摸事件处理
2. performSelector:onThread:
Source1具备唤醒线程的能力
1. 基于 Port 的线程间通信
2. 系统事件捕捉是由Source1来处理，然后再交给Source0处理
objc
// CFRunLoop.h
typedef struct __CFRunLoopSource * CFRunLoopSourceRef;
// CFRunLoop.m
struct __CFRunLoopSource {
    CFRuntimeBase _base;
    uint32_t _bits;
    pthread_mutex_t _lock;
    CFIndex _order;                       /* immutable */
    CFMutableBagRef _runLoops;
    union {
        CFRunLoopSourceContext version0;  // Source0
        CFRunLoopSourceContext1 version1; // Source1
    } _context;
};

2.2.4 CFRunLoopTimerRef ​

  1. CFRunloopTimer和NSTimer是toll-free bridged（可桥接）的
  2. performSelector:withObject:afterDelay:方法会创建timer并添加到RunLoop中
objc
// CFRunLoop.h
typedef struct CF_BRIDGED_MUTABLE_TYPE(NSTimer) __CFRunLoopTimer * CFRunLoopTimerRef;
// CFRunLoop.c
struct __CFRunLoopTimer {
    CFRuntimeBase _base;
    uint16_t _bits;
    pthread_mutex_t _lock;
    CFRunLoopRef _runLoop;           // 添加该 timer 的 RunLoop
    CFMutableSetRef _rlModes;        // 所有包含该 timer 的 modeName
    CFAbsoluteTime _nextFireDate;
    CFTimeInterval _interval;        /* immutable 理想时间间隔 */    
    CFTimeInterval _tolerance;       /* mutable 时间偏差 */  
    uint64_t _fireTSR;	             /* TSR units */
    CFIndex _order;                  /* immutable */
    CFRunLoopTimerCallBack _callout; /* immutable 回调入口 */
    CFRunLoopTimerContext _context;  /* immutable, except invalidation */
};

2.2.5 CFRunLoopObserverRef ​

  1. 用于监听RunLoop的状态
  2. UI刷新（kCFRunLoopBeforeWaiting）
  3. AutoreleasePool(kCFRunLoopBeforeWaiting)
objc
// 6种活动状态
typedef CF_OPTIONS(CFOptionFlags, CFRunLoopActivity) {
    kCFRunLoopEntry = (1UL << 0),          // 即将进入 RunLoop
    kCFRunLoopBeforeTimers = (1UL << 1),   // 即将处理 Timers
    kCFRunLoopBeforeSources = (1UL << 2),  // 即将处理 Sources
    kCFRunLoopBeforeWaiting = (1UL << 5),  // 即将进入休眠
    kCFRunLoopAfterWaiting = (1UL << 6),   // 刚从休眠中唤醒
    kCFRunLoopExit = (1UL << 7),           // 即将退出 RunLoop
    kCFRunLoopAllActivities = 0x0FFFFFFFU  // 表示以上所有状态
};

// CFRunLoop.h
typedef struct __CFRunLoopObserver * CFRunLoopObserverRef;
// CFRunLoop.c
struct __CFRunLoopObserver {
    CFRuntimeBase _base;
    pthread_mutex_t _lock;
    CFRunLoopRef _runLoop;              // 添加该 observer 的 RunLoop
    CFIndex _rlCount;
    CFOptionFlags _activities;	        /* immutable 监听的活动状态 */
    CFIndex _order;                     /* immutable */
    CFRunLoopObserverCallBack _callout;	/* immutable 回调入口 */
    CFRunLoopObserverContext _context;	/* immutable, except invalidation */
};

2.3 运行逻辑 ​

  1. 通知Observers：即将进入RunLoop
  2. 通知Observers：即将处理Timers
  3. 通知Observers：即将处理Sources
  4. 处理Blocks
  5. 处理Source0（可能会再次处理Blocks）
  6. 如果存在Source1，就跳转到第8步先处理Source1
  7. 通知Observers：开始休眠（等待消息唤醒）
  8. 通知Observers：结束休眠（被某个消息唤醒）
    1. 处理Timer
    2. 处理GCD Async To Main Queue
    3. 处理Source1
  9. 处理Blocks
  10. 如果还有Source、Timer、Observers等事件，就回到第02步通知Observers：即将处理Timers重新走流程
  11. 即将退出RunLoop
  12. 通知Observers：退出RunLoop
